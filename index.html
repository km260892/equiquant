    function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Trades');
  var data = sheet.getDataRange().getValues();
  var trades = [];
  
  for (var i = 1; i < data.length; i++) {
    trades.push({
      id: data[i][0],
      date: data[i][1],
      stock: data[i][2],
      type: data[i][3],
      entry: data[i][4],
      exit: data[i][5],
      quantity: data[i][6],
      pnl: data[i][7],
      status: data[i][8]
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify(trades))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Trades');
  var data = JSON.parse(e.postData.contents);
  
  if (data.action === 'add') {
    sheet.appendRow([
      data.id, data.date, data.stock, data.type,
      data.entry, data.exit, data.quantity, data.pnl, data.status
    ]);
  } else if (data.action === 'delete') {
    var values = sheet.getDataRange().getValues();
    for (var i = 1; i < values.length; i++) {
      if (values[i][0] == data.id) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({success: true}))
    .setMimeType(ContentService.MimeType.JSON);
}
